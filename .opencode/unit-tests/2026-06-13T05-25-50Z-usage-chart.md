import { describe, it, expect } from 'vitest'
import {
  mapUsageToChartData,
  mapUsageToModelData,
  type UsageRecordLike,
} from './usage-chart'

const baseUsage: UsageRecordLike[] = [
  // 2026-06-10 — 2 requests, 3.00 total
  { id: 'a1', model: 'gpt-4', promptTokens: 100, completionTokens: 50, costCents: '200', createdAt: '2026-06-10T10:00:00.000Z' },
  { id: 'a2', model: 'gpt-4', promptTokens: 100, completionTokens: 50, costCents: '100', createdAt: '2026-06-10T14:00:00.000Z' },
  // 2026-06-11 — 1 request, 0.50
  { id: 'b1', model: 'claude-3', promptTokens: 80, completionTokens: 40, costCents: '50', createdAt: '2026-06-11T09:00:00.000Z' },
  // 2026-06-12 — 3 requests, 1.25 total
  { id: 'c1', model: 'gpt-4', promptTokens: 200, completionTokens: 100, costCents: '75', createdAt: '2026-06-12T08:00:00.000Z' },
  { id: 'c2', model: 'claude-3', promptTokens: 200, completionTokens: 100, costCents: '25', createdAt: '2026-06-12T12:00:00.000Z' },
  { id: 'c3', model: 'claude-3', promptTokens: 200, completionTokens: 100, costCents: '25', createdAt: '2026-06-12T16:00:00.000Z' },
]

describe('usage-chart - Isolated Tests', () => {
  describe('mapUsageToChartData', () => {
    it('returns [] for empty input', () => {
      expect(mapUsageToChartData([])).toEqual([])
    })

    it('groups by day, sums cost, counts requests, returns last 7 days', () => {
      const result = mapUsageToChartData(baseUsage)
      expect(result).toEqual([
        { date: '2026-06-10', requests: 2, cost: 3.0 },
        { date: '2026-06-11', requests: 1, cost: 0.5 },
        { date: '2026-06-12', requests: 3, cost: 1.25 },
      ])
    })

    it('sorts chronologically even when input is out of order', () => {
      const shuffled = [
        baseUsage[4], // 06-12
        baseUsage[0], // 06-10
        baseUsage[2], // 06-11
      ]
      const result = mapUsageToChartData(shuffled)
      expect(result.map((d) => d.date)).toEqual(['2026-06-10', '2026-06-11', '2026-06-12'])
    })

    it('truncates to windowDays (e.g. windowDays=2 keeps only the last 2 days)', () => {
      const result = mapUsageToChartData(baseUsage, 2)
      expect(result).toEqual([
        { date: '2026-06-11', requests: 1, cost: 0.5 },
        { date: '2026-06-12', requests: 3, cost: 1.25 },
      ])
    })

    it('handles costCents as a number type too (not just string)', () => {
      const usage: UsageRecordLike[] = [
        { id: 'x1', model: 'gpt-4', promptTokens: 1, completionTokens: 1, costCents: 250, createdAt: '2026-06-13T10:00:00.000Z' },
      ]
      const result = mapUsageToChartData(usage)
      expect(result).toEqual([{ date: '2026-06-13', requests: 1, cost: 2.5 }])
    })

    it('uses ISO date (YYYY-MM-DD) regardless of the hour within the day', () => {
      const usage: UsageRecordLike[] = [
        { id: 'a', model: 'gpt-4', promptTokens: 0, completionTokens: 0, costCents: '100', createdAt: '2026-06-13T01:00:00.000Z' },
        { id: 'b', model: 'gpt-4', promptTokens: 0, completionTokens: 0, costCents: '200', createdAt: '2026-06-13T23:59:59.999Z' },
      ]
      const result = mapUsageToChartData(usage)
      expect(result).toEqual([{ date: '2026-06-13', requests: 2, cost: 3.0 }])
    })
  })

  describe('mapUsageToModelData', () => {
    it('returns [] for empty input', () => {
      expect(mapUsageToModelData([])).toEqual([])
    })

    it('aggregates cost per model, sorts desc, returns top 5 with palette colors', () => {
      const result = mapUsageToModelData(baseUsage)
      // gpt-4: 2.00 (a1) + 1.00 (a2) + 0.75 (c1) = 3.75
      // claude-3: 0.50 (b1) + 0.25 (c2) + 0.25 (c3) = 1.00
      expect(result).toEqual([
        { name: 'gpt-4', value: 3.75, color: expect.any(String) },
        { name: 'claude-3', value: 1.0, color: expect.any(String) },
      ])
      // Top entry gets the first palette color
      expect(result[0].color).toBe('#00ffc8') // CHART_PALETTE[0]
    })

    it('limits to topN=3 (e.g. when there are 4+ models)', () => {
      const usage: UsageRecordLike[] = [
        { id: '1', model: 'a', promptTokens: 0, completionTokens: 0, costCents: '100', createdAt: '2026-06-10T00:00:00.000Z' },
        { id: '2', model: 'b', promptTokens: 0, completionTokens: 0, costCents: '400', createdAt: '2026-06-10T00:00:00.000Z' },
        { id: '3', model: 'c', promptTokens: 0, completionTokens: 0, costCents: '300', createdAt: '2026-06-10T00:00:00.000Z' },
        { id: '4', model: 'd', promptTokens: 0, completionTokens: 0, costCents: '50', createdAt: '2026-06-10T00:00:00.000Z' },
      ]
      const result = mapUsageToModelData(usage, 3)
      expect(result).toHaveLength(3)
      expect(result.map((m) => m.name)).toEqual(['b', 'c', 'a']) // 4, 3, 2 cost
      expect(result.map((m) => m.value)).toEqual([4, 3, 1])
    })

    it('cycles through CHART_PALETTE for >8 models (the wrap-around case)', () => {
      const usage: UsageRecordLike[] = Array.from({ length: 10 }, (_, i) => ({
        id: String(i),
        model: `model-${i}`,
        promptTokens: 0,
        completionTokens: 0,
        costCents: String(100 - i),
        createdAt: '2026-06-10T00:00:00.000Z',
      }))
      const result = mapUsageToModelData(usage, 10)
      expect(result).toHaveLength(10)
      // First color = palette[0], 9th color = palette[8 % 8] = palette[0] again
      expect(result[0].color).toBe(result[8].color)
    })
  })
})
